附录 H 代码中访问控件

设计器负责生成布局和事件骨架；真正的业务代码需要按控件 ID 找到控件，读取数据、设置属性、绑定回调。本附录给出 C++ Binding 与 C ABI 两种写法。

设计器导出的 C++ 工程默认使用 C++ Binding（main.cpp）；导出的 C 工程使用 C ABI（main.c）。两套 API 访问的是同一棵控件树。

H.1 按 ID 找到控件

C++ Binding

#include "UICornerstone.h"
#include "Control.h"
#include "PropertyNames.h"
#include <cstdio>

// 假设 ui 已创建并加载了 layout.json
Control saveBtn = ui->FindControl("saveBtn");
if (!saveBtn.IsValid()) {
    std::fprintf(stderr, "找不到控件 saveBtn\n");
    return;
}

FindControl 返回 Control 代理。代理可拷贝，拷贝共享同一个控件状态；控件销毁后 IsValid() 会变为 false。

H.1 默认 ui / inst 已创建并加载布局，只聚焦“按 ID 查找”。完整 C++ 宿主见 H.11，完整 C ABI 宿主见 H.12。

C ABI

#include "UICornerstoneAPI.h"
#include <stdio.h>

UIControlHandle saveBtn = UICornerstone_FindControl(inst, "saveBtn");
if (!saveBtn) {
    fprintf(stderr, "找不到控件 saveBtn\n");
    return;
}

H.3 读取控件数据

C++ Binding

C++ Binding 的属性读取返回具体类型；属性名可以使用 PropertyNames.h 常量，也可以直接写字符串。

const std::string caption = saveBtn.GetString(PropertyNames::kCaption);
const bool visible       = saveBtn.GetBool(PropertyNames::kVisible);
const int selected       = saveBtn.GetInt(PropertyNames::kSelectedIndex);
const float fontSize     = saveBtn.GetFloat(PropertyNames::kFontSize);
const std::string anchor = saveBtn.GetEnum(PropertyNames::kAnchor);

UIColor bg = saveBtn.GetColor(PropertyNames::kBackground);
UIStateColor bgStates = saveBtn.GetStateColor(PropertyNames::kBackground);

UIRect r = saveBtn.GetRect();
float l = 0, t = 0, rr = 0, b = 0;
saveBtn.GetContentPadding(l, t, rr, b);
数据方法说明
文本GetString("caption") / GetString("text")Button/Label 用 caption；输入框用 text
整数GetInt("selected-index")索引、枚举数值
浮点GetFloat("font-size")字号、进度、范围
布尔GetBool("visible")可见、启用、勾选
颜色GetColor("background")普通单色
四态颜色GetStateColor("background")normal/hover/pressed/disabled
矩形GetRect()相对父容器的位置和尺寸
内容内边距GetContentPadding(...)content-padding
指针/对象GetPtr("...")行首控件等对象句柄，再用 FromHandle

C ABI

C ABI 的字符串读取需要缓冲区，并通过 outLen 得到完整长度；缓冲区不足时先查询长度再分配。

char caption[256] = {0};
int captionLen = 0;
if (UICornerstone_GetString(inst, saveBtn, "caption",
                            caption, sizeof(caption), &captionLen)) {
    printf("caption=%s len=%d\n", caption, captionLen);
}

int visible = 0;
int selected = -1;
float fontSize = 0.0f;
UICornerstone_GetBool(inst, saveBtn, "visible", &visible);
UICornerstone_GetInt(inst, saveBtn, "selected-index", &selected);
UICornerstone_GetFloat(inst, saveBtn, "font-size", &fontSize);

char anchor[64] = {0};
UICornerstone_GetEnum(inst, saveBtn, "anchor", anchor, sizeof(anchor));

UIColor bg;
UIStateColor bgStates;
if (UICornerstone_GetColor(inst, saveBtn, "background", &bg))
    printf("bg=%02X%02X%02X%02X\n", bg.r, bg.g, bg.b, bg.a);
if (UICornerstone_GetStateColor(inst, saveBtn, "background", &bgStates))
    printf("hover=%02X%02X%02X%02X\n",
           bgStates.hover.r, bgStates.hover.g,
           bgStates.hover.b, bgStates.hover.a);

float x = 0, y = 0, w = 0, h = 0;
float padL = 0, padT = 0, padR = 0, padB = 0;
UICornerstone_GetRect(inst, saveBtn, &x, &y, &w, &h);
UICornerstone_GetContentPadding(inst, saveBtn, &padL, &padT, &padR, &padB);
printf("rect=%.1f,%.1f %.1fx%.1f padding=%.1f,%.1f,%.1f,%.1f\n",
       x, y, w, h, padL, padT, padR, padB);

char type[64] = {0};
UICornerstone_GetControlType(inst, saveBtn, type, sizeof(type));
printf("id=%s type=%s\n", UICornerstone_GetControlId(inst, saveBtn), type);

如果需要读取完整字符串而缓冲区可能不足，先传 out=NULL 查询长度，再按 outLen 分配并读取。

H.5 绑定控件事件

C++ Binding

C++ Binding 用 SetCallback 注册事件回调，事件名使用运行时字符串（例如 "click"），不是 JSON 中的 "onClick"。

#include "Event.h"

saveBtn.SetCallback(PropertyNames::kEventClick, [this](const Event& e) {
    if (!e.IsClick()) return;
    std::printf("保存按钮被点击\n");
    doSave();
});
事件事件名常用读取
点击clickIsClick()
文本变化text-changedGetTextChanged()
数值变化value-changedGetValueChanged() / GetValueChangedDouble()
选择变化selection-changedGetSelectedIndex() / GetSelectedValue()
勾选变化check-changedGetCheckState()
页签切换tab-changedGetTabChangedIndex()
树节点选择selectGetNodeId()
列表选中list-selection-changedGetGridRow() / GetGridCol()
列表行点击item-clickGetGridRow() / GetGridCol()
单元格编辑完成cell-editedGetEditRow() / GetEditCol() / GetEditNewText()

解绑：

saveBtn.SetCallback(PropertyNames::kEventClick, {});   // 空回调 = 解绑

C ABI 事件回调

C ABI 通过 UICornerstone_SetCallback 注册，回调签名是 void (*)(UIControlHandle, const UIEventData*, void*)。事件负载在 ev->data union 中，按事件名读取。

#include <string.h>

typedef struct {
    UIInstance inst;
    int saveCount;
    int saved;
} AppState;

static void on_control_event(UIControlHandle ctl, const UIEventData* ev, void* user) {
    AppState* app = (AppState*)user;
    if (!ev || !ev->eventName) return;

    if (strcmp(ev->eventName, "click") == 0) {
        app->saveCount++;
        printf("click: %s\n", UICornerstone_GetControlId(app->inst, ctl));
    } else if (strcmp(ev->eventName, "text-changed") == 0) {
        printf("text: %s\n", ev->data.strVal ? ev->data.strVal : "");
    } else if (strcmp(ev->eventName, "value-changed") == 0) {
        printf("value: %f / %f\n", ev->data.floatVal, ev->data.doubleVal);
    } else if (strcmp(ev->eventName, "selection-changed") == 0) {
        printf("selection: %d / %s\n", ev->data.selection.idx,
               ev->data.selection.val ? ev->data.selection.val : "");
    } else if (strcmp(ev->eventName, "item-click") == 0) {
        printf("item: %d,%d\n", ev->data.grid.row, ev->data.grid.col);
    } else if (strcmp(ev->eventName, "check-changed") == 0) {
        printf("check: %d\n", ev->data.intVal);
    } else if (strcmp(ev->eventName, "tab-changed") == 0) {
        printf("tab: %d\n", ev->data.intVal);
    } else if (strcmp(ev->eventName, "select") == 0) {
        printf("tree node: %s\n", ev->data.treeNode.id ? ev->data.treeNode.id : "");
    } else if (strcmp(ev->eventName, "list-selection-changed") == 0) {
        printf("list: %d,%d\n", ev->data.grid.row, ev->data.grid.col);
    } else if (strcmp(ev->eventName, "cell-edited") == 0) {
        printf("edited: %d,%d = %s\n", ev->data.grid.row, ev->data.grid.col,
               ev->data.grid.newText ? ev->data.grid.newText : "");
    } else if (strcmp(ev->eventName, "color-changed") == 0) {
        printf("color: %02X%02X%02X%02X\n", ev->data.color.r, ev->data.color.g,
               ev->data.color.b, ev->data.color.a);
    }
}

UIControlHandle nameEdit = UICornerstone_FindControl(inst, "nameEdit");
UIControlHandle listView = UICornerstone_FindControl(inst, "userList");

AppState app = { inst, 0, 0 };
UICornerstone_SetCallback(inst, saveBtn, "click", on_control_event, &app);
UICornerstone_SetCallback(inst, nameEdit, "text-changed", on_control_event, &app);
UICornerstone_SetCallback(inst, listView, "list-selection-changed", on_control_event, &app);
事件C ABI 负载
click无
text-changeddata.strVal
value-changeddata.floatVal 或 data.doubleVal
selection-changeddata.selection.idx / data.selection.val
check-changeddata.intVal
tab-changeddata.intVal
selectdata.treeNode.id / data.treeNode.userData
list-selection-changeddata.grid.row / data.grid.col
cell-editeddata.grid.row/col/newText
color-changeddata.color

解绑：

UICornerstone_SetCallback(inst, saveBtn, "click", NULL, NULL);

H.6 事件回调中的控件访问

C++ Binding

回调的第一个参数通常是事件源；也可以按 ID 再查找其他控件：

nameEdit.SetCallback(PropertyNames::kEventTextChanged, [ui = ui.get()](const Event& e) {
    const std::string text = e.GetTextChanged();
    auto status = ui->FindControl("statusLabel");
    if (status.IsValid()) status.SetString(PropertyNames::kCaption, "你好，" + text);
});

回调在 ProcessEvents 期间执行，应在 UI 线程运行。不要在回调里做长时间阻塞操作；重活可以记录状态，延到下一帧处理。

LoadLayout 重建控件树后，之前缓存的 Control 代理或裸句柄可能失效。重建后重新 FindControl，并在使用前检查 IsValid()。

C ABI 回调中访问其他控件

typedef struct {
    UIInstance inst;
} TextState;

static void on_name_changed(UIControlHandle ctl, const UIEventData* ev, void* user) {
    (void)ctl;
    TextState* app = (TextState*)user;
    UIControlHandle status = UICornerstone_FindControl(app->inst, "statusLabel");
    if (!status) return;

    char msg[512];
    snprintf(msg, sizeof(msg), "你好，%s", ev->data.strVal ? ev->data.strVal : "");
    UICornerstone_SetString(app->inst, status, "caption", msg);
}

UIControlHandle nameEdit = UICornerstone_FindControl(inst, "nameEdit");
TextState app = { inst };
UICornerstone_SetCallback(inst, nameEdit, "text-changed", on_name_changed, &app);

C ABI 回调同样在 ProcessEvents 中执行；耗时工作应延后，不要阻塞事件泵。

H.6.1 运行时切换回调

运行中改变事件处理逻辑，优先使用控件级 SetCallback，不要依赖对已加载布局重新 RegisterAction。有两种稳定做法：

做法 A：保留一个稳定的分发回调，只切换内部状态/函数对象。这种方式不会反复改引擎绑定，最适合按登录态、编辑态、权限等切换业务逻辑。

class Page {
public:
    void Bind() {
        m_saveBtn = m_ui->FindControl("saveBtn");
        m_saveBtn.SetCallback(PropertyNames::kEventClick, [this](const Event& e) {
            if (m_onSave) m_onSave(e);   // 每次点击读取当前逻辑
        });
    }

    void SetAdminMode(bool admin) {
        m_onSave = admin
            ? std::function<void(const Event&)>([this](const Event&) { adminSave(); })
            : std::function<void(const Event&)>([this](const Event&) { normalSave(); });
    }

private:
    UICornerstone::UICornerstone* m_ui = nullptr;
    Control m_saveBtn;
    std::function<void(const Event&)> m_onSave;
};

做法 B：直接替换回调。找到控件后再次调用 SetCallback，新回调会替换该控件该事件的引擎绑定；旧回调不再被调用。

auto saveBtn = ui->FindControl("saveBtn");
if (saveBtn.IsValid()) {
    saveBtn.SetCallback(PropertyNames::kEventClick, [](const Event&) {
        // 新的点击逻辑
    });
}

C ABI 同样使用稳定的 C 函数指针分发，或直接再次调用 UICornerstone_SetCallback 替换：

typedef void (*SaveHandler)(UIControlHandle, const UIEventData*, void*);

static void on_admin_save(UIControlHandle ctl, const UIEventData* ev, void* user) {
    (void)ctl; (void)ev; (void)user;
    /* 管理员模式逻辑 */
}

static void on_normal_save(UIControlHandle ctl, const UIEventData* ev, void* user) {
    (void)ctl; (void)ev; (void)user;
    /* 普通模式逻辑 */
}

static SaveHandler g_saveHandler = NULL;
static void dispatch_save(UIControlHandle ctl, const UIEventData* ev, void* user) {
    if (g_saveHandler) g_saveHandler(ctl, ev, user);
}

/* 初始化时只绑定一次 */
UICornerstone_SetCallback(inst, saveBtn, "click", dispatch_save, &app);

/* 运行中切换逻辑 */
g_saveHandler = on_admin_save;   // 或 on_normal_save

/* 或者直接替换该控件该事件的回调 */
UICornerstone_SetCallback(inst, saveBtn, "click", on_admin_save, &app);
RegisterAction 是布局解析期的动作表：运行时再次注册同名 action 主要影响后续 LoadLayout，不保证改变已经解析并绑定的控件。要在运行中立即切换，使用 SetCallback；确实要换 JSON action，请重新加载布局，但注意控件树会重建、旧句柄/代理可能失效。

回调在 UI 线程的 ProcessEvents 中执行；如果切换逻辑来自其他线程，请先把状态安全地传到 UI 线程，不要在工作线程直接调用 SetCallback 或修改控件。

H.7 JSON 动作与 SetCallback 的区别

C++ Binding

方式适合注册
SetCallback("click", ...)代码直接控制某个控件；可以拿到 Event 负载控件创建/找到之后
RegisterAction("onSave", ...)布局 JSON 的 events: {"onClick":"onSave"}LoadLayout 之前注册；回调只拿控件，不拿 Event
// 布局 JSON: "events": { "onClick": "onSave" }
ui->RegisterAction("onSave", [this](Control ctl) {
    std::printf("action=%s\n", ctl.GetId().c_str());
});
ui->LoadLayoutFromFile("layout.json");
为什么必须在 LoadLayout 之前注册？ RegisterAction 只是登记“动作名 → 回调”表；LoadLayout 创建布局解析器时，才把这张表转成 parser handler，并在解析每个控件的 events 字段时按名字查找。若加载时还没登记，解析器会静默跳过该事件；之后再调用 RegisterAction 不会回头扫描并绑定已经创建的控件。

如果确实需要“先加载布局、后绑定”，请改用控件级 SetCallback（可以拿到 Event 负载），或注册后重新加载布局。重新加载会重建控件树，之前缓存的句柄/代理可能失效。

设计器导出的 C++ 工程使用 RegisterAction 把事件名映射到 events.c 中的函数：

ui->RegisterAction("onSaveClick", [](Control c) {
    onSaveClick(c.Handle(), nullptr);
});

如果需要在代码中读取事件负载（文本、行号、颜色），请对控件使用 SetCallback；如果只是响应设计器事件面板里填写的 handler，使用 RegisterAction。

C ABI 的 RegisterAction

C ABI 的动作回调签名是 void (*)(UIControlHandle, void*)，同样必须在 LoadLayout 之前注册：

static void on_save_action(UIControlHandle ctl, void* user) {
    AppState* app = (AppState*)user;
    app->saved = 1;
    printf("action control id=%s\n",
           UICornerstone_GetControlId(app->inst, ctl));
}

AppState app = { inst, 0, 0 };
UICornerstone_RegisterAction(inst, "onSaveClick", on_save_action, &app);
UICornerstone_LoadLayoutFromFile(inst, "layout.json");

H.8 C ABI 事件回调

下面把 SetCallback 与 RegisterAction 放在同一个 C ABI 场景中：on_save_click 处理控件点击，on_save_action 响应 JSON 动作。

/* 复用 H.5 的 AppState：inst / saveCount / saved */

static void on_save_click(UIControlHandle ctl, const UIEventData* ev, void* user) {
    (void)ev;
    (void)ctl;
    AppState* app = (AppState*)user;
    app->saved = 1;
}

AppState appState = { inst, 0, 0 };
UIControlHandle saveBtn = UICornerstone_FindControl(inst, "saveBtn");
UICornerstone_SetCallback(inst, saveBtn, "click", on_save_click, &appState);

JSON 动作使用另一种回调签名：

static void on_save_action(UIControlHandle ctl, void* user) {
    (void)ctl;
    (void)user;
}

UICornerstone_RegisterAction(inst, "onSaveClick", on_save_action, &appState);

H.11 最小完整示例

C++ Binding

#include "UICornerstone.h"
#include "Control.h"
#include "Event.h"
#include "PropertyNames.h"
#include <cstdio>

int main() {
    auto ui = UICornerstone::UICornerstone::Create(
        UICornerstone::UICornerstone::Config{}.WithBackend("sdl3"));

    ui->RegisterAction("onSaveClick", [](Control c) {
        std::printf("action from %s\n", c.GetId().c_str());
    });
    ui->LoadLayoutFromFile("layout.json");

    Control name = ui->FindControl("nameEdit");
    Control status = ui->FindControl("statusLabel");
    if (name.IsValid() && status.IsValid()) {
        name.SetCallback(PropertyNames::kEventTextChanged,
            [status](const Event& e) mutable {
                status.SetString(PropertyNames::kCaption, "你好，" + e.GetTextChanged());
            });
    }

    while (!ui->IsQuitRequested()) {
        ui->ProcessEvents();
        ui->Update(0.016);
        ui->Clear();
        ui->Render();
        ui->Present();
    }
}

C ABI 最小完整示例

#include "UICornerstoneAPI.h"
#include <stdio.h>
#include <string.h>

typedef struct {
    UIInstance inst;
    UIControlHandle status;
} App;

static void on_save_action(UIControlHandle ctl, void* user) {
    (void)ctl;
    (void)user;
    printf("onSaveClick action\n");
}

static void on_name_changed(UIControlHandle ctl, const UIEventData* ev, void* user) {
    (void)ctl;
    App* app = (App*)user;
    char msg[512];
    snprintf(msg, sizeof(msg), "你好，%s", ev->data.strVal ? ev->data.strVal : "");
    UICornerstone_SetString(app->inst, app->status, "caption", msg);
}

int main(void) {
    UIInstanceConfig cfg = UI_INSTANCE_CONFIG_DEFAULT;
    cfg.windowTitle = "C ABI Demo";
    cfg.windowWidth = 1024;
    cfg.windowHeight = 768;
    cfg.resourceRoot = "assets";

    UIInstance inst = UICornerstone_CreateInstanceFromPlugin("sdl3", &cfg);
    if (!inst) return 1;

    UICornerstone_RegisterAction(inst, "onSaveClick", on_save_action, NULL);
    if (!UICornerstone_LoadLayoutFromFile(inst, "layout.json")) {
        UICornerstone_DestroyInstance(inst);
        return 1;
    }

    UIControlHandle nameEdit = UICornerstone_FindControl(inst, "nameEdit");
    UIControlHandle status   = UICornerstone_FindControl(inst, "statusLabel");
    if (!nameEdit || !status) {
        UICornerstone_DestroyInstance(inst);
        return 1;
    }

    App app = { inst, status };
    UICornerstone_SetCallback(inst, nameEdit, "text-changed", on_name_changed, &app);

    while (!UICornerstone_IsQuitRequested(inst)) {
        UICornerstone_ProcessEvents(inst);
        UICornerstone_Update(inst, 1.0 / 60.0);
        UICornerstone_Clear(inst);
        UICornerstone_Render(inst);
        UICornerstone_Present(inst);
    }

    UICornerstone_DestroyInstance(inst);
    return 0;
}

H.12 C ABI 完整宿主骨架

H.1 只演示“拿到控件句柄”；这一节给出 C ABI 的完整宿主：创建实例、注册 JSON 动作、加载布局、查找控件和帧循环。实际导出的 C 工程通过 LoadLibrary/GetProcAddress 显式解析同一组函数，避免依赖导入库；函数调用方式完全一致。

#include "UICornerstoneAPI.h"
#include <stdio.h>
#include <string.h>

static UIInstance g_inst = NULL;
static int g_saved = 0;

static void on_save_action(UIControlHandle ctl, void* user) {
    (void)user;
    g_saved = 1;
    printf("action control id=%s\n", UICornerstone_GetControlId(g_inst, ctl));
}

int main(void) {
    UIInstanceConfig cfg = UI_INSTANCE_CONFIG_DEFAULT;
    cfg.windowTitle  = "Cornerstone C Host";
    cfg.windowWidth  = 1280;
    cfg.windowHeight = 800;
    cfg.windowFlags  = UICORN_WINDOW_FLAG_RESIZABLE;
    cfg.resourceRoot = "assets";

    UIInstance inst = UICornerstone_CreateInstanceFromPlugin("sdl3", &cfg);
    if (!inst) { fprintf(stderr, "创建实例失败\n"); return 1; }
    g_inst = inst;

    /* JSON 动作必须在 LoadLayout 之前注册 */
    UICornerstone_RegisterAction(inst, "onSaveClick", on_save_action, NULL);

    if (!UICornerstone_LoadLayoutFromFile(inst, "layout.json")) {
        fprintf(stderr, "加载 layout.json 失败\n");
        UICornerstone_DestroyInstance(inst);
        return 1;
    }

    UIControlHandle saveBtn = UICornerstone_FindControl(inst, "saveBtn");
    if (!saveBtn) {
        fprintf(stderr, "找不到控件 saveBtn\n");
        UICornerstone_DestroyInstance(inst);
        return 1;
    }

    while (!UICornerstone_IsQuitRequested(inst)) {
        UICornerstone_ProcessEvents(inst);
        UICornerstone_Update(inst, 1.0 / 60.0);
        UICornerstone_Clear(inst);
        UICornerstone_Render(inst);
        UICornerstone_Present(inst);
    }

    UICornerstone_DestroyInstance(inst);
    return 0;
}
