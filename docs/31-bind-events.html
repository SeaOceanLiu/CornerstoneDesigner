31 事件回调方式 2：绑定控件事件

C++ Binding

C++ Binding 用 SetCallback 注册事件回调，事件名使用运行时字符串（例如 "click"），不是 JSON 中的 "onClick"。

#include "Event.h"

saveBtn.SetCallback(PropertyNames::kEventClick, [this](const Event& e) {
    if (!e.IsClick()) return;
    std::printf("保存按钮被点击\n");
    doSave();
});
事件事件名常用读取
点击clickIsClick()
文本变化text-changedGetTextChanged()
数值变化value-changedGetValueChanged() / GetValueChangedDouble()
选择变化selection-changedGetSelectedIndex() / GetSelectedValue()
勾选变化check-changedGetCheckState()
页签切换tab-changedGetTabChangedIndex()
树节点选择selectGetNodeId()
列表选中list-selection-changedGetGridRow() / GetGridCol()
列表行点击item-clickGetGridRow() / GetGridCol()
单元格编辑完成cell-editedGetEditRow() / GetEditCol() / GetEditNewText()

解绑：

saveBtn.SetCallback(PropertyNames::kEventClick, {});   // 空回调 = 解绑

C ABI 事件回调

C ABI 通过 UICornerstone_SetCallback 注册，回调签名是 void (*)(UIControlHandle, const UIEventData*, void*)。事件负载在 ev->data union 中，按事件名读取。

#include <string.h>

typedef struct {
    UIInstance inst;
    int saveCount;
    int saved;
} AppState;

static void on_control_event(UIControlHandle ctl, const UIEventData* ev, void* user) {
    AppState* app = (AppState*)user;
    if (!ev || !ev->eventName) return;

    if (strcmp(ev->eventName, "click") == 0) {
        app->saveCount++;
        printf("click: %s\n", UICornerstone_GetControlId(app->inst, ctl));
    } else if (strcmp(ev->eventName, "text-changed") == 0) {
        printf("text: %s\n", ev->data.strVal ? ev->data.strVal : "");
    } else if (strcmp(ev->eventName, "value-changed") == 0) {
        printf("value: %f / %f\n", ev->data.floatVal, ev->data.doubleVal);
    } else if (strcmp(ev->eventName, "selection-changed") == 0) {
        printf("selection: %d / %s\n", ev->data.selection.idx,
               ev->data.selection.val ? ev->data.selection.val : "");
    } else if (strcmp(ev->eventName, "item-click") == 0) {
        printf("item: %d,%d\n", ev->data.grid.row, ev->data.grid.col);
    } else if (strcmp(ev->eventName, "check-changed") == 0) {
        printf("check: %d\n", ev->data.intVal);
    } else if (strcmp(ev->eventName, "tab-changed") == 0) {
        printf("tab: %d\n", ev->data.intVal);
    } else if (strcmp(ev->eventName, "select") == 0) {
        printf("tree node: %s\n", ev->data.treeNode.id ? ev->data.treeNode.id : "");
    } else if (strcmp(ev->eventName, "list-selection-changed") == 0) {
        printf("list: %d,%d\n", ev->data.grid.row, ev->data.grid.col);
    } else if (strcmp(ev->eventName, "cell-edited") == 0) {
        printf("edited: %d,%d = %s\n", ev->data.grid.row, ev->data.grid.col,
               ev->data.grid.newText ? ev->data.grid.newText : "");
    } else if (strcmp(ev->eventName, "color-changed") == 0) {
        printf("color: %02X%02X%02X%02X\n", ev->data.color.r, ev->data.color.g,
               ev->data.color.b, ev->data.color.a);
    }
}

UIControlHandle nameEdit = UICornerstone_FindControl(inst, "nameEdit");
UIControlHandle listView = UICornerstone_FindControl(inst, "userList");

AppState app = { inst, 0, 0 };
UICornerstone_SetCallback(inst, saveBtn, "click", on_control_event, &app);
UICornerstone_SetCallback(inst, nameEdit, "text-changed", on_control_event, &app);
UICornerstone_SetCallback(inst, listView, "list-selection-changed", on_control_event, &app);
事件C ABI 负载
click无
text-changeddata.strVal
value-changeddata.floatVal 或 data.doubleVal
selection-changeddata.selection.idx / data.selection.val
check-changeddata.intVal
tab-changeddata.intVal
selectdata.treeNode.id / data.treeNode.userData
list-selection-changeddata.grid.row / data.grid.col
cell-editeddata.grid.row/col/newText
color-changeddata.color

解绑：

UICornerstone_SetCallback(inst, saveBtn, "click", NULL, NULL);
